// ── PRODUCTO INDIVIDUAL ───────────────────────────────────────
function cvaGetProducto(clave, opts) {
  // Vista detallada de producto individual — todos los parámetros.
  // Solo se llama cuando el usuario abre un producto específico, no en búsquedas de lista.
  if (!clave) throw new Error("Se requiere clave CVA");

  const params = {
    clave,
    MonedaPesos : opts.MonedaPesos  || "true",
    porcentaje  : opts.porcentaje   || CFG.MARGEN_DEFAULT,
    promos      : opts.promos       || "true",
    images      : opts.images       || "true",
    tc          : opts.tc           || "true",
    sucursales  : opts.sucursales   || "true",
    dimen       : opts.dimen        || "true",
    dt          : opts.dt           || "true",
    dc          : opts.dc           || "true",
    upc         : opts.upc          || "true",
    codigosat   : opts.codigosat    || "true",
    TipoProducto: opts.TipoProducto || "true",
    tipo        : opts.tipo         || "true",
    depto       : opts.depto        || "true",
    trans       : opts.trans        || "true",
  };

  const data = cvaFetch_("/catalogo_clientes/lista_precios", params);
  return { ok: true, producto: data };
}

// ── BÚSQUEDA / FILTROS ────────────────────────────────────────
function cvaSearch(opts) {
  const params = {};

  if (opts.marca)    params.marca    = opts.marca;
  if (opts.grupo)    params.grupo    = opts.grupo;
  if (opts.desc)     params.desc     = opts.desc;
  if (opts.grupo2)   params.grupo2   = opts.grupo2;
  if (opts.Solucion) params.Solucion = opts.Solucion;
  if (opts.exist && opts.exist !== "any") params.exist = opts.exist;

  params.batch = opts.batch || "SM";
  params.page  = opts.page  || 1;

  params.MonedaPesos  = "true";
  params.porcentaje   = opts.porcentaje || CFG.MARGEN_DEFAULT;
  params.tc           = "true";
  // NO mandamos promos/sucursales/dimen/dt/dc/upc/codigosat en búsquedas de lista
  // — esos parámetros triplican el tiempo de respuesta de CVA
  // Solo se piden en cvaGetProducto (vista individual)
  if (opts.promos)    params.promos    = "true";
  if (opts.images)    params.images    = "true";
  if (opts.completos) { params.images  = "true"; params.completos = "true"; }

  const data = cvaFetch_("/catalogo_clientes/lista_precios", params);

  if (opts.exist === "any" && data.articulos) {
    data.articulos = data.articulos.filter(a =>
      (parseFloat(a.disponible) || 0) > 0 || (parseFloat(a.disponibleCD) || 0) > 0
    );
  }

  return { ok: true, ...data };
}

// ── CREAR ORDEN CVA ───────────────────────────────────────────
function cvaCrearOrden(body) {
  if (!body) throw new Error("body es undefined — error de parsing en doPost");
  const { productos, direccion, num_oc, tipo_flete, test, observaciones, codigo_sucursal } = body;

  if (!productos || productos.length === 0) {
    throw new Error("Se requiere al menos 1 producto");
  }

  const orderBody = {
    num_oc         : num_oc || "",
    codigo_sucursal: (codigo_sucursal !== undefined && codigo_sucursal !== null)
                     ? parseInt(codigo_sucursal) : CFG.CVA_SUCURSAL,
    tipo_flete     : tipo_flete || "SF",
    observaciones  : observaciones || "Dropship Electronics México",
    cotiza_flete   : 1,
    MonedaPesos    : true,   // facturar en MXN
    moneda         : "MXN",  // forzar moneda MXN en el pedido
    productos      : productos.map(p => ({
      clave   : p.clave,
      cantidad: parseInt(p.cantidad) || 1,
    })),
  };

  if (test) orderBody.test = 1;

  if (tipo_flete !== "SF" && direccion) {
    orderBody.flete = {
      calle           : direccion.calle    || "",
      numero          : direccion.numero   || "S/N",
      numero_interior : direccion.interior || "",
      cp              : direccion.cp       || "",
      colonia         : direccion.colonia  || "",
      estado          : parseInt(direccion.estado_id),
      ciudad          : parseInt(direccion.ciudad_id),
      paqueteria      : parseInt(direccion.paqueteria_id) || 4,
      atencion        : direccion.atencion || "Cliente",
    };
  }

  const data = cvaPost_("/pedidos_web/crear_orden?MonedaPesos=true", orderBody);

  logSheet_("PEDIDOS_CVA", [
    num_oc || "",
    data.pedido || "",
    data.total || "",
    data.moneda || "",
    tipo_flete || "FF",
    test ? "TEST" : "REAL",
    JSON.stringify(productos),
  ]);

  return { ok: true, ...data };
}

// ── COTIZAR FLETE ─────────────────────────────────────────────
function cvaCotizarFlete(cp_destino, cp_sucursal, productos) {
  const res = UrlFetchApp.fetch("https://www.grupocva.com/api/paqueteria_service/", {
    method: "post",
    contentType: "application/json",
    payload: JSON.stringify({
      paqueteria : 4,
      cp         : parseInt(cp_destino),
      cp_sucursal: parseInt(cp_sucursal) || 44900,
      productos  : productos.map(p => ({
        clave   : p.clave,
        cantidad: parseInt(p.cantidad) || 1,
      })),
    }),
    muteHttpExceptions: true,
  });

  const data = JSON.parse(res.getContentText());
  return { ok: true, ...data };
}

// ── SYNC MASIVO CVA → LOCAL ───────────────────────────────────
// OPTIMIZACIÓN: primero consulta cuántas páginas tiene el catálogo con
// una petición batch=SM page=1, y usa ese total para evitar 404s.
// Auto-reset si CVA devuelve 404 o "No se encontraron productos".
function syncPreciosCVAOdoo(opts) {
  const batch   = opts.batch   || "XL";
  const maxPag  = opts.paginas || 999;
  const START_MS = Date.now();
  const MAX_MS   = 5 * 60 * 1000; // 5 min

  const props = PropertiesService.getScriptProperties();

  // Honrar reset pendiente
  if (props.getProperty("SYNC_RESET_FLAG") === "true") {
    props.deleteProperty("SYNC_RESET_FLAG");
    props.setProperty("SYNC_PAGE", "1");
    props.deleteProperty("SYNC_TOTAL_PAGINAS"); // forzar re-consulta del total
  }

  let pagina = parseInt(props.getProperty("SYNC_PAGE") || "1");

  // Guardar/leer filtro de exist
  if (opts.exist !== undefined && opts.exist !== null) {
    props.setProperty("SYNC_EXIST", String(opts.exist));
  }
  const existFilter = props.getProperty("SYNC_EXIST") || "3";

  // ── OPTIMIZACIÓN: leer total de páginas guardado o consultarlo una vez ──
  // Guardamos el total para no perder el ciclo si el catálogo cambia de tamaño
  let totalPaginas = parseInt(props.getProperty("SYNC_TOTAL_PAGINAS") || "0");

  if (totalPaginas === 0 || pagina === 1) {
    // Consultar total real con una petición ligera (batch SM = 50 art)
    try {
      const probe = cvaFetch_("/catalogo_clientes/lista_precios", {
        batch: "SM", page: 1,
        MonedaPesos: "true",
        porcentaje: CFG.MARGEN_DEFAULT,
        exist: existFilter || "3",
      });
      totalPaginas = probe.paginacion ? probe.paginacion.total_paginas : 999;
      props.setProperty("SYNC_TOTAL_PAGINAS", String(totalPaginas));
      Logger.log("📊 Total páginas CVA: " + totalPaginas);
    } catch(e) {
      totalPaginas = 999; // si falla la probe, seguir con valor alto
    }
  }

  let totalArticulos = 0;

  const ss = SpreadsheetApp.openById(CFG.SHEET_ID);
  let sh = ss.getSheetByName("SYNC_CVA");
  if (!sh) {
    sh = ss.insertSheet("SYNC_CVA");
    sh.appendRow(["clave","descripcion","marca","grupo","precio","moneda",
                  "stock_suc","stock_cedis","en_transito","garantia",
                  "promo_precio","promo_vence","tipo_cambio","ts"]);
    sh.setFrozenRows(1);
  }

  for (let i = 0; i < maxPag; i++) {

    // Guard: tiempo
    if (Date.now() - START_MS > MAX_MS) {
      logSheet_("SYNC_LOG", ["SYNC_TIMEOUT", totalArticulos, "next_page:" + pagina]);
      _actualizarControlSync_(props, pagina, totalArticulos, "TIMEOUT");
      break;
    }

    // Guard: reset externo
    if (props.getProperty("SYNC_RESET_FLAG") === "true") {
      props.deleteProperty("SYNC_RESET_FLAG");
      props.setProperty("SYNC_PAGE", "1");
      props.deleteProperty("SYNC_TOTAL_PAGINAS");
      logSheet_("SYNC_LOG", ["SYNC_RESET", totalArticulos, "reset externo"]);
      _actualizarControlSync_(props, 1, totalArticulos, "RESET");
      return { ok: true, articulos_procesados: totalArticulos, next_page: 1, reseteado: true };
    }

    // Guard: fin del catálogo — reiniciar ciclo
    if (pagina > totalPaginas) {
      props.setProperty("SYNC_PAGE", "1");
      props.deleteProperty("SYNC_TOTAL_PAGINAS"); // refrescar total en próximo ciclo
      logSheet_("SYNC_LOG", ["SYNC_COMPLETO", totalArticulos, "páginas:" + totalPaginas]);
      _actualizarControlSync_(props, 1, totalArticulos, "COMPLETO");
      break;
    }

    // ── Fetch página ──────────────────────────────────────────
    const fetchParams = {
      batch, page: pagina,
      MonedaPesos: "true",
      porcentaje : CFG.MARGEN_DEFAULT,
      tc         : "true",
      sucursales : "true",
      trans      : "true",
      promos     : "true",
    };
    if (existFilter !== "") fetchParams.exist = existFilter;

    let data;
    try {
      data = cvaFetch_("/catalogo_clientes/lista_precios", fetchParams);
    } catch (fetchErr) {
      const msg = fetchErr.message || "";
      // 404 o "No se encontraron" = checkpoint fuera de rango → auto-reset
      if (msg.includes("404") || msg.includes("No se encontraron")) {
        Logger.log("⚠️ Página " + pagina + " fuera de rango — auto-reset a 1");
        props.setProperty("SYNC_PAGE", "1");
        props.deleteProperty("SYNC_TOTAL_PAGINAS");
        logSheet_("SYNC_LOG", ["SYNC_RESET_AUTO", totalArticulos,
          "pág " + pagina + " no existe (total=" + totalPaginas + ") — reiniciando"]);
        _actualizarControlSync_(props, 1, totalArticulos, "AUTO-RESET");
        return { ok: true, articulos_procesados: totalArticulos, next_page: 1,
                 total_paginas: totalPaginas, auto_reset: true };
      }
      // Otro error (timeout, red) — loguear y salir sin perder el checkpoint
      logSheet_("SYNC_LOG", ["SYNC_ERROR", totalArticulos, msg.substring(0, 100)]);
      _actualizarControlSync_(props, pagina, totalArticulos, "ERROR");
      throw fetchErr;
    }

    // CVA devuelve 200 pero con mensaje de "No se encontraron productos"
    if (!data || data.message === "No se encontraron productos." ||
        (!data.articulos && !data.paginacion)) {
      Logger.log("⚠️ Sin productos en pág " + pagina + " — auto-reset");
      props.setProperty("SYNC_PAGE", "1");
      props.deleteProperty("SYNC_TOTAL_PAGINAS");
      logSheet_("SYNC_LOG", ["SYNC_RESET_AUTO", totalArticulos,
        "sin productos pág " + pagina + " — reiniciando"]);
      _actualizarControlSync_(props, 1, totalArticulos, "AUTO-RESET");
      return { ok: true, articulos_procesados: totalArticulos, next_page: 1,
               total_paginas: totalPaginas, auto_reset: true };
    }

    // Actualizar total si cambió
    if (data.paginacion && data.paginacion.total_paginas !== totalPaginas) {
      totalPaginas = data.paginacion.total_paginas;
      props.setProperty("SYNC_TOTAL_PAGINAS", String(totalPaginas));
    }

    // ── Filtrar y guardar artículos ───────────────────────────
    const articulos = data.articulos || [];
    const articulosFiltrados = articulos.filter(a =>
      (parseFloat(a.disponible) || 0) > 0 || (parseFloat(a.disponibleCD) || 0) > 0
    );

    const ts   = new Date().toISOString();
    const fecha = ts.substring(0, 10);

    // UPSERT en SYNC_CVA
    const lastRow = sh.getLastRow();
    let claveIndex = {};
    if (lastRow > 1) {
      const claves = sh.getRange(2, 1, lastRow - 1, 1).getValues();
      claves.forEach((r, idx) => { if (r[0]) claveIndex[r[0]] = idx + 2; });
    }

    const rowsToAppend = [];
    articulosFiltrados.forEach(a => {
      const promo = a.promociones || a.promocion || null;
      const row = [
        a.clave        || "",
        a.descripcion  || a.codigo || "",
        a.marca        || "",
        a.grupo        || "",
        a.precio       || 0,
        a.moneda       || "Pesos",
        a.disponible   || 0,
        a.disponibleCD || 0,
        a.en_transito  || 0,
        a.garantia     || "",
        promo ? (promo.precio_descuento || promo.precio || "") : "",
        promo ? (promo.promocion_vencimiento || promo.vencimiento || "") : "",
        a.tipo_cambio  || "",
        ts,
      ];
      const existingRow = claveIndex[a.clave];
      if (existingRow) {
        sh.getRange(existingRow, 1, 1, 14).setValues([row]);
      } else {
        rowsToAppend.push(row);
      }
    });
    if (rowsToAppend.length > 0) {
      sh.getRange(sh.getLastRow() + 1, 1, rowsToAppend.length, 14).setValues(rowsToAppend);
    }

    // HISTORIAL_STOCK: snapshot por hora
    let shH = ss.getSheetByName("HISTORIAL_STOCK");
    if (!shH) {
      shH = ss.insertSheet("HISTORIAL_STOCK");
      shH.appendRow(["fecha","hora","clave","descripcion","marca","precio",
                     "moneda","stock_suc","stock_cedis","en_transito"]);
      shH.setFrozenRows(1);
    }

    const horaActual = ts.substring(0, 13);
    const lastHRow   = shH.getLastRow();
    let   lastHora   = "";
    if (lastHRow > 1) {
      const lastTs       = String(shH.getRange(lastHRow, 1).getValue());
      const lastHoraVal  = String(shH.getRange(lastHRow, 2).getValue());
      lastHora = lastTs.substring(0, 10) + "T" + lastHoraVal.substring(0, 2);
    }

    if (lastHora !== horaActual && articulosFiltrados.length > 0) {
      const hora = ts.substring(11, 13) + ":00";
      const histRows = articulosFiltrados.map(a => [
        fecha, hora,
        a.clave        || "",
        a.descripcion  || a.codigo || "",
        a.marca        || "",
        a.precio       || 0,
        a.moneda       || "Pesos",
        a.disponible   || 0,
        a.disponibleCD || 0,
        a.en_transito  || 0,
      ]);
      shH.getRange(shH.getLastRow() + 1, 1, histRows.length, 10).setValues(histRows);
    }

    totalArticulos += articulosFiltrados.length;
    pagina++;
  }

  props.setProperty("SYNC_PAGE", String(pagina));
  logSheet_("SYNC_LOG", ["SYNC_PARCIAL", totalArticulos, "next_page:" + pagina,
    "total_pags:" + totalPaginas]);
  _actualizarControlSync_(props, pagina, totalArticulos, "OK");

  return {
    ok: true,
    articulos_procesados: totalArticulos,
    next_page: pagina,
    total_paginas: totalPaginas,
  };
}

// Actualiza la hoja CONTROL_SYNC con el último estado del sync
function _actualizarControlSync_(props, pagina, articulos, estado) {
  try {
    const ss = SpreadsheetApp.openById(CFG.SHEET_ID);
    let sh = ss.getSheetByName("CONTROL_SYNC");
    if (!sh) return; // la hoja la crea sheets_menu.gs
    const now = new Date();
    // Columna B2:B7 = valores de estado
    sh.getRange("B2").setValue(now);
    sh.getRange("B3").setValue(estado);
    sh.getRange("B4").setValue(pagina);
    sh.getRange("B5").setValue(parseInt(props.getProperty("SYNC_TOTAL_PAGINAS") || "0"));
    sh.getRange("B6").setValue(articulos);
    sh.getRange("B7").setValue(
      ss.getSheetByName("SYNC_CVA") ? ss.getSheetByName("SYNC_CVA").getLastRow() - 1 : 0
    );
  } catch(e) {
    Logger.log("_actualizarControlSync_ error: " + e.message);
  }
}

// ── GUÍAS DE ENVÍO ───────────────────────────────────────────
const CVA_GUIAS_URL = "https://www.grupocva.com/api/v1/orders/guias/";

function cvaEnviarGuia(body) {
  const { order_number, waybills, carrier, pdf_base64 } = body;

  if (!order_number) throw new Error("Se requiere order_number");
  if (!waybills)     throw new Error("Se requiere waybill(s)");
  if (!carrier)      throw new Error("Se requiere carrier");

  const token = CFG.CVA_GUIAS_TOKEN;

  const res = UrlFetchApp.fetch(CVA_GUIAS_URL, {
    method: "post",
    contentType: "application/json",
    headers: { Authorization: "Bearer " + token },
    payload: JSON.stringify({ order_number, waybills, carrier, file: pdf_base64 || "" }),
    muteHttpExceptions: true,
  });

  const code = res.getResponseCode();
  const text = res.getContentText();
  if (code !== 200) throw new Error("CVA Guías HTTP " + code + ": " + text);

  const data = JSON.parse(text);
  if (data.result !== 0 && data.result !== "0") {
    throw new Error("CVA Guías error: " + (data.message || JSON.stringify(data)));
  }

  try { actualizarGuiaEnSheet_(order_number, waybills, carrier); } catch(e) {}
  logSheet_("SYNC_LOG", ["GUIA_ENVIADA", order_number, waybills, carrier]);

  return { ok: true, result: data.result, order_number, waybills };
}

function cvaConsultarGuia(order_number) {
  if (!order_number) throw new Error("Se requiere order_number");
  const token = CFG.CVA_GUIAS_TOKEN;

  const res = UrlFetchApp.fetch(CVA_GUIAS_URL + "?order_number=" + encodeURIComponent(order_number), {
    method: "get",
    headers: { Authorization: "Bearer " + token },
    muteHttpExceptions: true,
  });

  const code = res.getResponseCode();
  if (code !== 200) throw new Error("CVA Guías GET HTTP " + code + ": " + res.getContentText());

  const data = JSON.parse(res.getContentText());
  return { ok: true, ...data };
}

function actualizarGuiaEnSheet_(order_number, waybills, carrier) {
  const ss = SpreadsheetApp.openById(CFG.SHEET_ID);
  const sh = ss.getSheetByName("PEDIDOS_GUIAS");
  if (!sh) return;

  const data    = sh.getDataRange().getValues();
  const headers = data[0];
  const colOC   = headers.indexOf("orden_cva");
  const colGuia = headers.indexOf("guia_enviada");
  const colWB   = headers.indexOf("no_guia");

  for (let i = 1; i < data.length; i++) {
    if (String(data[i][colOC]) === String(order_number)) {
      sh.getRange(i + 1, colGuia + 1).setValue("SI");
      sh.getRange(i + 1, colWB   + 1).setValue(waybills);
      break;
    }
  }
}

// ── POLLING PEDIDOS PENDIENTES ────────────────────────────────
function pollingPedidosCVA() {
  const ss = SpreadsheetApp.openById(CFG.SHEET_ID);
  const sh = ss.getSheetByName("PEDIDOS_CVA");
  if (!sh) return;

  const data         = sh.getDataRange().getValues();
  const colPedidoCVA = 1;
  const colEstatus   = 7;
  let uid;
  try { uid = autenticarXMLRPC_(); } catch(e) { uid = null; }

  for (let i = 1; i < data.length; i++) {
    const row       = data[i];
    const pedidoCVA = String(row[colPedidoCVA] || "").trim();
    const estatus   = String(row[colEstatus]   || "").trim();

    if (!pedidoCVA || estatus === "FACTURADO" || estatus === "CANCELADO") continue;

    try {
      const res          = cvaConsultarPedido(pedidoCVA);
      const nuevoEstatus = res.pedido?.estatus || "PENDIENTE";
      sh.getRange(i + 1, colEstatus + 1).setValue(nuevoEstatus);

      if (nuevoEstatus === "FACTURADO") {
        const factura = res.pedido?.factura || "";
        sh.getRange(i + 1, colEstatus + 2).setValue(factura);
        logSheet_("SYNC_LOG", ["PEDIDO_FACTURADO", pedidoCVA, factura]);
        const numOC = String(row[0] || "");
        if (numOC && uid) actualizarEstatusPOOdoo_(uid, numOC, "purchase", factura);
      }
    } catch(e) {
      Logger.log("Polling error " + pedidoCVA + ": " + e.message);
    }
  }
